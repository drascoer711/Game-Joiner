import os
import aiohttp
import discord
from discord import app_commands
from dotenv import load_dotenv

load_dotenv()

TOKEN = os.getenv("DISCORD_TOKEN")

if not TOKEN:
    raise RuntimeError("DISCORD_TOKEN is missing from .env")


USERS_API = "https://users.roblox.com/v1"
GROUPS_API = "https://groups.roblox.com/v2"
THUMBNAILS_API = "https://thumbnails.roblox.com/v1"
BADGES_API = "https://badges.roblox.com/v1"


# -----------------------------
# Roblox API helper
# -----------------------------

async def request(method, url, **kwargs):
    try:
        async with aiohttp.ClientSession() as session:
            async with session.request(
                method,
                url,
                timeout=15,
                **kwargs
            ) as response:

                if response.status != 200:
                    return None

                return await response.json()

    except Exception:
        return None


# -----------------------------
# Find Roblox user
# -----------------------------

async def find_user(username):

    data = await request(
        "POST",
        f"{USERS_API}/usernames/users",
        json={
            "usernames": [username],
            "excludeBannedUsers": False
        }
    )

    if not data or not data.get("data"):
        return None

    return data["data"][0]


# -----------------------------
# User information
# -----------------------------

async def get_user(user_id):

    return await request(
        "GET",
        f"{USERS_API}/users/{user_id}"
    )


# -----------------------------
# Avatar
# -----------------------------

async def get_avatar(user_id):

    data = await request(
        "GET",
        f"{THUMBNAILS_API}/users/avatar-headshot",
        params={
            "userIds": user_id,
            "size": "420x420",
            "format": "Png",
            "isCircular": "false"
        }
    )

    if not data or not data.get("data"):
        return None

    return data["data"][0].get("imageUrl")


# -----------------------------
# Groups
# -----------------------------

async def get_groups(user_id):

    data = await request(
        "GET",
        f"{GROUPS_API}/users/{user_id}/groups/roles"
    )

    if not data:
        return []

    return data.get("data", [])


# -----------------------------
# Badges
# -----------------------------

async def get_badges(user_id):

    data = await request(
        "GET",
        f"{BADGES_API}/users/{user_id}/badges",
        params={
            "limit": 10,
            "sortOrder": "Desc"
        }
    )

    if not data:
        return []

    return data.get("data", [])


# -----------------------------
# Calculate possible match
# -----------------------------

def calculate_match(
    target_name,
    candidate_name,
    shared_groups
):

    score = 0

    target = target_name.lower()
    candidate = candidate_name.lower()

    if target == candidate:
        score += 30

    elif target in candidate or candidate in target:
        score += 20

    if shared_groups >= 5:
        score += 35

    elif shared_groups >= 3:
        score += 25

    elif shared_groups >= 1:
        score += 10

    return min(score, 99)

# Discord bot
# -----------------------------

class RobloxBot(discord.Client):

    def __init__(self):

        intents = discord.Intents.default()

        super().__init__(
            intents=intents
        )

        self.tree = app_commands.CommandTree(self)

        # Voluntarily provided country information
        self.user_countries = {}

    async def setup_hook(self):

        await self.tree.sync()

        print("Slash commands synced.")


bot = RobloxBot()


# =========================================================
# /user
# =========================================================

@bot.tree.command(
    name="user",
    description="Search for a Roblox user."
)
@app_commands.describe(
    username="Roblox username"
)
async def user(
    interaction: discord.Interaction,
    username: str
):

    await interaction.response.defer()

    user = await find_user(username)

    if not user:

        await interaction.followup.send(
            f"❌ Roblox user `{username}` was not found."
        )

        return

    user_id = user["id"]

    info = await get_user(user_id)

    if not info:

        await interaction.followup.send(
            "❌ Couldn't retrieve Roblox information."
        )

        return

    avatar = await get_avatar(user_id)

    embed = discord.Embed(
        title="🔎 Roblox User",
        color=discord.Color.blurple()
    )

    embed.add_field(
        name="Username",
        value=f"`{info.get('name', 'Unknown')}`",
        inline=True
    )

    embed.add_field(
        name="Display Name",
        value=info.get("displayName", "Unknown"),
        inline=True
    )

    embed.add_field(
        name="User ID",
        value=f"`{user_id}`",
        inline=True
    )

    embed.add_field(
        name="Created",
        value=info.get("created", "Unknown")[:10],
        inline=True
    )

    embed.add_field(
        name="Description",
        value=info.get("description") or "No description",
        inline=False
    )

    if avatar:
        embed.set_thumbnail(url=avatar)

    embed.add_field(
        name="Profile",
        value=f"https://www.roblox.com/users/{user_id}/profile",
        inline=False
    )

    await interaction.followup.send(
        embed=embed
    )

@bot.tree.command(
    name="groups",
    description="Show a Roblox user's groups."
)
@app_commands.describe(
    username="Roblox username"
)
async def groups(
    interaction: discord.Interaction,
    username: str
):

    await interaction.response.defer()

    user = await find_user(username)

    if not user:

        await interaction.followup.send(
            "❌ User not found."
        )

        return

    groups = await get_groups(user["id"])

    embed = discord.Embed(
        title=f"👥 Groups — {user['name']}",
        color=discord.Color.blurple()
    )

    if not groups:

        embed.description = "No groups found."

    else:

        lines = []

        for group in groups[:20]:

            group_info = group["group"]
            role = group["role"]

            lines.append(
                f"**{group_info['name']}**\n"
                f"Role: `{role['name']}`\n"
                f"ID: `{group_info['id']}`"
            )

        embed.description = "\n\n".join(lines)

    embed.set_footer(
        text=f"{len(groups)} groups found"
    )

    await interaction.followup.send(
        embed=embed
    )
